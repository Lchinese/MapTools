# MapTools 数据库设计文档

## 概述

本文档详细描述了 MapTools 系统的数据库设计，包括表结构、索引、关系约束和存储过程。数据库采用 MySQL 8.0，并启用了空间扩展以支持地理空间数据存储和查询。

## 数据库配置

### 基本信息
- **数据库类型**: MySQL 8.0
- **字符集**: utf8mb4
- **排序规则**: utf8mb4_unicode_ci
- **存储引擎**: InnoDB
- **空间参考系统**: SRID 4490 (中国2000坐标系)

### 连接配置
```sql
-- 创建数据库
CREATE DATABASE maptools 
CHARACTER SET utf8mb4 
COLLATE utf8mb4_unicode_ci;

-- 创建用户
CREATE USER 'maptools_user'@'%' IDENTIFIED BY 'maptools_password';
GRANT ALL PRIVILEGES ON maptools.* TO 'maptools_user'@'%';
FLUSH PRIVILEGES;
```

## 表结构设计

### 1. 用户表 (users)

存储系统用户信息。

```sql
CREATE TABLE users (
    user_id VARCHAR(36) PRIMARY KEY COMMENT '用户ID',
    username VARCHAR(50) NOT NULL UNIQUE COMMENT '用户名',
    email VARCHAR(100) NOT NULL UNIQUE COMMENT '邮箱',
    password_hash VARCHAR(255) NOT NULL COMMENT '密码哈希',
    full_name VARCHAR(100) COMMENT '全名',
    avatar_url VARCHAR(500) COMMENT '头像URL',
    is_active BOOLEAN DEFAULT TRUE COMMENT '是否激活',
    is_admin BOOLEAN DEFAULT FALSE COMMENT '是否管理员',
    last_login_at TIMESTAMP NULL COMMENT '最后登录时间',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    
    INDEX idx_username (username),
    INDEX idx_email (email),
    INDEX idx_created_at (created_at)
) ENGINE=InnoDB COMMENT='用户表';
```

### 2. 轨迹表 (trajectories)

存储轨迹基本信息。

```sql
CREATE TABLE trajectories (
    trajectory_id VARCHAR(36) PRIMARY KEY COMMENT '轨迹ID',
    user_id VARCHAR(36) NOT NULL COMMENT '用户ID',
    name VARCHAR(200) NOT NULL COMMENT '轨迹名称',
    description TEXT COMMENT '轨迹描述',
    filename VARCHAR(255) NOT NULL COMMENT '原始文件名',
    file_path VARCHAR(500) NOT NULL COMMENT '文件存储路径',
    file_size BIGINT NOT NULL COMMENT '文件大小（字节）',
    file_type ENUM('gpx', 'kml', 'csv', 'txt') NOT NULL COMMENT '文件类型',
    data_source ENUM('gpx', 'taxi_gps', 'bus_card', 'metro_card', 'taxi_transaction', 'bus_gps', 'csv') NOT NULL COMMENT '数据源类型',
    data_category ENUM('continuous_trajectory', 'origin_destination', 'time_range') NOT NULL COMMENT '数据类别',
    point_count INT NOT NULL DEFAULT 0 COMMENT '轨迹点数量',
    status ENUM('uploaded', 'processing', 'completed', 'failed') DEFAULT 'uploaded' COMMENT '处理状态',
    error_message TEXT COMMENT '错误信息',
    
    -- 空间边界框
    min_latitude DECIMAL(10, 8) COMMENT '最小纬度',
    max_latitude DECIMAL(10, 8) COMMENT '最大纬度',
    min_longitude DECIMAL(11, 8) COMMENT '最小经度',
    max_longitude DECIMAL(11, 8) COMMENT '最大经度',
    
    -- 时间范围
    start_time TIMESTAMP NULL COMMENT '开始时间',
    end_time TIMESTAMP NULL COMMENT '结束时间',
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    
    FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE,
    INDEX idx_user_id (user_id),
    INDEX idx_status (status),
    INDEX idx_created_at (created_at),
    INDEX idx_file_type (file_type),
    SPATIAL INDEX idx_bounds (min_latitude, max_latitude, min_longitude, max_longitude)
) ENGINE=InnoDB COMMENT='轨迹表';
```

### 3. 轨迹点表 (trajectory_points)

存储轨迹的详细点数据。

```sql
CREATE TABLE trajectory_points (
    point_id VARCHAR(36) PRIMARY KEY COMMENT '点ID',
    trajectory_id VARCHAR(36) NOT NULL COMMENT '轨迹ID',
    sequence_number INT NOT NULL COMMENT '序列号',
    
    -- 坐标信息
    latitude DECIMAL(10, 8) NOT NULL COMMENT '纬度',
    longitude DECIMAL(11, 8) NOT NULL COMMENT '经度',
    elevation DECIMAL(8, 2) COMMENT '海拔（米）',
    
    -- 空间点（使用 SRID 4490）
    location POINT SRID 4490 COMMENT '空间点坐标',
    
    -- 时间信息
    timestamp TIMESTAMP NOT NULL COMMENT '时间戳',
    
    -- 精度信息
    accuracy DECIMAL(8, 2) COMMENT '精度（米）',
    speed DECIMAL(8, 2) COMMENT '速度（米/秒）',
    bearing DECIMAL(5, 2) COMMENT '方向角（度）',
    
    -- 原始数据
    raw_data JSON COMMENT '原始数据（JSON格式）',
    
    -- 车辆/乘客信息
    vehicle_id VARCHAR(50) COMMENT '车辆标识（车牌号等）',
    passenger_id VARCHAR(50) COMMENT '乘客标识（卡号等）',
    line_id VARCHAR(50) COMMENT '线路标识',
    station_id VARCHAR(50) COMMENT '站点标识',
    
    -- 业务数据
    speed DECIMAL(8, 2) COMMENT '速度（公里/小时）',
    direction DECIMAL(5, 2) COMMENT '方向角（度）',
    fare DECIMAL(8, 2) COMMENT '费用（分）',
    transaction_type VARCHAR(10) COMMENT '交易类型',
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    
    FOREIGN KEY (trajectory_id) REFERENCES trajectories(trajectory_id) ON DELETE CASCADE,
    INDEX idx_trajectory_id (trajectory_id),
    INDEX idx_sequence (trajectory_id, sequence_number),
    INDEX idx_timestamp (timestamp),
    SPATIAL INDEX idx_location (location)
) ENGINE=InnoDB COMMENT='轨迹点表';
```

### 4. 地图匹配任务表 (matching_tasks)

存储地图匹配任务信息。

```sql
CREATE TABLE matching_tasks (
    task_id VARCHAR(36) PRIMARY KEY COMMENT '任务ID',
    trajectory_id VARCHAR(36) NOT NULL COMMENT '轨迹ID',
    user_id VARCHAR(36) NOT NULL COMMENT '用户ID',
    
    -- 算法配置
    algorithm ENUM('hmm', 'greedy') NOT NULL COMMENT '匹配算法',
    algorithm_parameters JSON COMMENT '算法参数',
    road_network_id VARCHAR(50) DEFAULT 'default' COMMENT '路网ID',
    
    -- 任务状态
    status ENUM('queued', 'running', 'completed', 'failed', 'cancelled') DEFAULT 'queued' COMMENT '任务状态',
    progress INT DEFAULT 0 COMMENT '进度百分比',
    error_message TEXT COMMENT '错误信息',
    
    -- 结果统计
    total_points INT DEFAULT 0 COMMENT '总点数',
    matched_points INT DEFAULT 0 COMMENT '匹配点数',
    unmatched_points INT DEFAULT 0 COMMENT '未匹配点数',
    accuracy DECIMAL(5, 2) COMMENT '匹配精度',
    processing_time DECIMAL(8, 2) COMMENT '处理时间（秒）',
    
    -- 时间信息
    queued_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '入队时间',
    started_at TIMESTAMP NULL COMMENT '开始时间',
    completed_at TIMESTAMP NULL COMMENT '完成时间',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    
    FOREIGN KEY (trajectory_id) REFERENCES trajectories(trajectory_id) ON DELETE CASCADE,
    FOREIGN KEY (user_id) REFERENCES users(user_id) ON DELETE CASCADE,
    INDEX idx_trajectory_id (trajectory_id),
    INDEX idx_user_id (user_id),
    INDEX idx_status (status),
    INDEX idx_algorithm (algorithm),
    INDEX idx_created_at (created_at)
) ENGINE=InnoDB COMMENT='地图匹配任务表';
```

### 5. 匹配结果表 (matching_results)

存储地图匹配的详细结果。

```sql
CREATE TABLE matching_results (
    result_id VARCHAR(36) PRIMARY KEY COMMENT '结果ID',
    task_id VARCHAR(36) NOT NULL COMMENT '任务ID',
    trajectory_id VARCHAR(36) NOT NULL COMMENT '轨迹ID',
    point_id VARCHAR(36) NOT NULL COMMENT '原始点ID',
    
    -- 匹配后的坐标
    matched_latitude DECIMAL(10, 8) NOT NULL COMMENT '匹配后纬度',
    matched_longitude DECIMAL(11, 8) NOT NULL COMMENT '匹配后经度',
    matched_location POINT SRID 4490 COMMENT '匹配后空间点',
    
    -- 匹配信息
    road_id VARCHAR(100) COMMENT '匹配道路ID',
    road_name VARCHAR(200) COMMENT '道路名称',
    confidence DECIMAL(5, 4) COMMENT '匹配置信度',
    distance DECIMAL(8, 2) COMMENT '匹配距离（米）',
    
    -- 原始坐标（冗余存储）
    original_latitude DECIMAL(10, 8) NOT NULL COMMENT '原始纬度',
    original_longitude DECIMAL(11, 8) NOT NULL COMMENT '原始经度',
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    
    FOREIGN KEY (task_id) REFERENCES matching_tasks(task_id) ON DELETE CASCADE,
    FOREIGN KEY (trajectory_id) REFERENCES trajectories(trajectory_id) ON DELETE CASCADE,
    FOREIGN KEY (point_id) REFERENCES trajectory_points(point_id) ON DELETE CASCADE,
    INDEX idx_task_id (task_id),
    INDEX idx_trajectory_id (trajectory_id),
    INDEX idx_point_id (point_id),
    INDEX idx_road_id (road_id),
    SPATIAL INDEX idx_matched_location (matched_location)
) ENGINE=InnoDB COMMENT='匹配结果表';
```

## 索引设计

### 空间索引
```sql
-- 轨迹点空间索引
CREATE SPATIAL INDEX idx_trajectory_points_location ON trajectory_points(location);

-- 匹配结果空间索引
CREATE SPATIAL INDEX idx_matching_results_location ON matching_results(matched_location);

-- 道路几何空间索引
CREATE SPATIAL INDEX idx_roads_geometry ON roads(geometry);

-- 路网覆盖范围空间索引
CREATE SPATIAL INDEX idx_road_networks_bounds ON road_networks(coverage_bounds);
```

### 复合索引
```sql
-- 轨迹查询优化
CREATE INDEX idx_trajectories_user_status ON trajectories(user_id, status);
CREATE INDEX idx_trajectories_created_status ON trajectories(created_at, status);

-- 轨迹点查询优化
CREATE INDEX idx_trajectory_points_traj_seq ON trajectory_points(trajectory_id, sequence_number);
CREATE INDEX idx_trajectory_points_timestamp ON trajectory_points(timestamp);

-- 匹配任务查询优化
CREATE INDEX idx_matching_tasks_user_status ON matching_tasks(user_id, status);
CREATE INDEX idx_matching_tasks_traj_status ON matching_tasks(trajectory_id, status);

-- 匹配结果查询优化
CREATE INDEX idx_matching_results_task_point ON matching_results(task_id, point_id);
```

### 1. 创建轨迹存储过程

```sql
DELIMITER //

    IN p_file_path VARCHAR(500),
    IN p_file_size BIGINT,
    IN p_file_type ENUM('gpx', 'kml', 'csv'),
    OUT p_trajectory_id VARCHAR(36)
)
BEGIN
    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        RESIGNAL;
    END;
    
    START TRANSACTION;
    
    -- 生成轨迹ID
    SET p_trajectory_id = UUID();
    
    -- 插入轨迹记录
    INSERT INTO trajectories (
        trajectory_id, user_id, name, description, filename, 
        file_path, file_size, file_type, status
    ) VALUES (
        p_trajectory_id, p_user_id, p_name, p_description, p_filename,
        p_file_path, p_file_size, p_file_type, 'uploaded'
    );
    
    COMMIT;
END //
DELIMITER ;
```

### 2. 更新轨迹边界框存储过程

```sql
DELIMITER //
CREATE PROCEDURE UpdateTrajectoryBounds(
    IN p_trajectory_id VARCHAR(36)
)
BEGIN
    DECLARE v_min_lat, v_max_lat DECIMAL(10, 8);
    DECLARE v_min_lng, v_max_lng DECIMAL(11, 8);
    DECLARE v_start_time, v_end_time TIMESTAMP;
    
    -- 计算边界框
    SELECT 
        MIN(latitude), MAX(latitude),
        MIN(longitude), MAX(longitude),
        MIN(timestamp), MAX(timestamp)
    INTO v_min_lat, v_max_lat, v_min_lng, v_max_lng, v_start_time, v_end_time
    FROM trajectory_points 
    WHERE trajectory_id = p_trajectory_id;
    
    -- 更新轨迹表
    UPDATE trajectories 
    SET 
        min_latitude = v_min_lat,
        max_latitude = v_max_lat,
        min_longitude = v_min_lng,
        max_longitude = v_max_lng,
        start_time = v_start_time,
        end_time = v_end_time,
        point_count = (
            SELECT COUNT(*) FROM trajectory_points 
            WHERE trajectory_id = p_trajectory_id
        )
    WHERE trajectory_id = p_trajectory_id;
    
END //
DELIMITER ;
```

### 3. 清理过期数据存储过程

```sql
DELIMITER //
CREATE PROCEDURE CleanupExpiredData(
    IN p_days_to_keep INT DEFAULT 30
)
BEGIN
    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        RESIGNAL;
    END;
    
    START TRANSACTION;
    
    -- 删除过期的失败任务
    DELETE FROM matching_tasks 
    WHERE status = 'failed' 
    AND created_at < DATE_SUB(NOW(), INTERVAL p_days_to_keep DAY);
    
    -- 删除过期的系统日志
    DELETE FROM system_logs 
    WHERE created_at < DATE_SUB(NOW(), INTERVAL p_days_to_keep DAY);
    
    -- 删除孤立的文件记录
    DELETE FROM files 
    WHERE trajectory_id IS NULL 
    AND created_at < DATE_SUB(NOW(), INTERVAL p_days_to_keep DAY);
    
    COMMIT;
END //
DELIMITER ;
```

## 视图定义

### 1. 轨迹统计视图

```sql
CREATE VIEW trajectory_statistics AS
SELECT 
    t.trajectory_id,
    t.name,
    t.user_id,
    u.username,
    t.point_count,
    t.status,
    t.created_at,
    t.updated_at,
    CASE 
        WHEN t.start_time IS NOT NULL AND t.end_time IS NOT NULL 
        THEN TIMESTAMPDIFF(SECOND, t.start_time, t.end_time)
        ELSE NULL
    END AS duration_seconds,
    CASE 
        WHEN t.min_latitude IS NOT NULL AND t.max_latitude IS NOT NULL 
        THEN ST_Distance_Sphere(
            POINT(t.min_longitude, t.min_latitude),
            POINT(t.max_longitude, t.max_latitude)
        )
        ELSE NULL
    END AS bounding_distance
FROM trajectories t
JOIN users u ON t.user_id = u.user_id;
```

### 2. 匹配任务统计视图

```sql
CREATE VIEW matching_task_statistics AS
SELECT 
    mt.task_id,
    mt.trajectory_id,
    t.name AS trajectory_name,
    mt.user_id,
    u.username,
    mt.algorithm,
    mt.status,
    mt.total_points,
    mt.matched_points,
    mt.unmatched_points,
    mt.accuracy,
    mt.processing_time,
    mt.created_at,
    mt.completed_at,
    CASE 
        WHEN mt.started_at IS NOT NULL AND mt.completed_at IS NOT NULL 
        THEN TIMESTAMPDIFF(SECOND, mt.started_at, mt.completed_at)
        ELSE NULL
    END AS actual_processing_time
FROM matching_tasks mt
JOIN trajectories t ON mt.trajectory_id = t.trajectory_id
JOIN users u ON mt.user_id = u.user_id;
```
